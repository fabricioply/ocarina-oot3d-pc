cmake_minimum_required(VERSION 3.20)
project(RealInstrument LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_library(real_instrument src/pitch_detector.cpp src/note_tracker.cpp src/song_matcher.cpp src/song_catalog.cpp src/session.cpp src/game_input.cpp src/mic_settings.cpp)
target_include_directories(real_instrument PUBLIC include)
if(MSVC)
  target_compile_options(real_instrument PRIVATE /W4 /permissive-)
else()
  target_compile_options(real_instrument PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(test_pitch tests/test_pitch.cpp)
target_link_libraries(test_pitch PRIVATE real_instrument)
add_test(NAME pitch COMMAND test_pitch)
add_executable(test_notes tests/test_notes.cpp)
target_link_libraries(test_notes PRIVATE real_instrument)
add_test(NAME notes COMMAND test_notes)
add_executable(test_matcher tests/test_matcher.cpp)
target_link_libraries(test_matcher PRIVATE real_instrument)
add_test(NAME matcher COMMAND test_matcher)
add_executable(test_session tests/test_session.cpp)
target_link_libraries(test_session PRIVATE real_instrument)
add_test(NAME session COMMAND test_session)
add_executable(test_game_input tests/test_game_input.cpp)
target_link_libraries(test_game_input PRIVATE real_instrument)
add_test(NAME game_input COMMAND test_game_input)
add_executable(test_oot3d_state tests/test_oot3d_state.cpp)
target_link_libraries(test_oot3d_state PRIVATE real_instrument)
add_test(NAME oot3d_state COMMAND test_oot3d_state)
add_executable(test_mic_settings tests/test_mic_settings.cpp)
target_link_libraries(test_mic_settings PRIVATE real_instrument)
add_test(NAME mic_settings COMMAND test_mic_settings)
if(WIN32)
  add_library(real_instrument_capture src/wasapi_capture.cpp)
  target_include_directories(real_instrument_capture PUBLIC src)
  target_compile_definitions(real_instrument_capture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(real_instrument_capture PUBLIC ole32 uuid)
  add_library(real_instrument_game_microphone src/game_microphone.cpp src/mic_controller.cpp)
  target_link_libraries(real_instrument_game_microphone PUBLIC real_instrument real_instrument_capture)
  if(MSVC)
    target_compile_options(real_instrument_game_microphone PRIVATE /W4 /permissive-)
  endif()
  add_executable(test_game_microphone tests/test_game_microphone.cpp)
  target_link_libraries(test_game_microphone PRIVATE real_instrument_game_microphone)
  add_test(NAME game_microphone_lifecycle COMMAND test_game_microphone)
  add_executable(test_mic_controller tests/test_mic_controller.cpp)
  target_link_libraries(test_mic_controller PRIVATE real_instrument_game_microphone)
  add_test(NAME mic_controller COMMAND test_mic_controller)
  if(MSVC)
    target_compile_options(real_instrument_capture PRIVATE /W4 /permissive-)
  endif()
  add_executable(test_capture tests/test_capture.cpp)
  target_link_libraries(test_capture PRIVATE real_instrument_capture)
  add_test(NAME capture_invalid_device COMMAND test_capture)
  add_executable(OcarinaMicrofone src/main.cpp)
  target_link_libraries(OcarinaMicrofone PRIVATE real_instrument real_instrument_capture)
  target_compile_definitions(OcarinaMicrofone PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(OcarinaMicrofone PRIVATE /W4 /permissive-)
  endif()
endif()
